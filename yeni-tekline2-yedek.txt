# Matthews Correlation Coefficient (MCC) — caret doğrudan vermez
calc_mcc <- function(cm) {
  tp <- cm$table[1, 1]  # Predicted Mutant, Actual Mutant
  tn <- cm$table[2, 2]  # Predicted WT, Actual WT
  fp <- cm$table[1, 2]  # Predicted Mutant, Actual WT
  fn <- cm$table[2, 1]  # Predicted WT, Actual Mutant
  denom <- sqrt(as.numeric(tp + fp) * as.numeric(tp + fn) *
                as.numeric(tn + fp) * as.numeric(tn + fn))
  if (denom == 0) return(0)
  return((tp * tn - fp * fn) / denom)
}

remove_probes_step2 <- unique(c(snp_probe_maf, snp_cpg_rs, snp_sbe_rs, rs_probes))
meth_step2 <- meth_step1[!(rownames(meth_step1) %in% remove_probes_step2), ]
cat(sprintf("    Çıkarılan prob: %d | Kalan: %d\n",
            nrow(meth_step1) - nrow(meth_step2), nrow(meth_step2)))
rm(meth_step1, snp_probe_maf, snp_cpg_rs, snp_sbe_rs, rs_probes, remove_probes_step2)

  chen_url <- "https://raw.githubusercontent.com/sirselim/illumina450k_crossreactive/master/450k_crossreactive.csv"
  cr_probes <- data.table::fread(chen_url, header = FALSE)$V1
  cat(sprintf("    Chen listesi indirildi: %d cross-reactive prob\n", length(cr_probes)))
}, error = function(e) {
  cat("    Hata: Chen listesi indirilemedi. Bu adım atlanıyor.\n")
})

if (!is.null(cr_probes)) {
  meth_step3 <- meth_step2[!(rownames(meth_step2) %in% cr_probes), ]
  cat(sprintf("    Çıkarılan prob: %d | Kalan: %d\n",
              nrow(meth_step2) - nrow(meth_step3), nrow(meth_step3)))
} else {
  meth_step3 <- meth_step2
  cat("    Cross-reactive filtreleme atlandı.\n")

na_frac    <- rowSums(is.na(meth_step3)) / ncol(meth_step3)
meth_step4 <- meth_step3[na_frac <= 0.20, ]
cat(sprintf("    Çıkarılan prob: %d | Kalan: %d\n",
            nrow(meth_step3) - nrow(meth_step4), nrow(meth_step4)))
rm(meth_step3, na_frac); gc()

# ---- Adım 5: MAD top 5000 — kNN'den ÖNCE (RAM Optimizasyonu) ----

# MAD hesabı NA'lı matris üzerinde na.rm=TRUE ile yapılır; ardından 5000 prob seçilir.
# kNN imputation 5000×595 üzerinde çalışır → ~64 kat daha küçük matris.
# Not: NA içeren probları komşu seçimi biraz etkiler, ancak bu ölçekte kabul edilebilir.
cat("  Adım 5: MAD top 5000 prob seçimi uygulanıyor (kNN öncesi, na.rm=TRUE)...\n")
mad_meth   <- apply(meth_step4, 1, mad, na.rm = TRUE)
top5k_meth <- names(sort(mad_meth, decreasing = TRUE))[1:5000]
meth_filt4 <- meth_step4[top5k_meth, ]
cat(sprintf("    Seçilen prob sayısı: %d\n", nrow(meth_filt4)))
rm(meth_step4, mad_meth, top5k_meth); gc()

# ---- Adım 6: Kalan NA → kNN Imputation (5000 prob — RAM uyumlu) ----
cat("  Adım 6: Kalan NA'lar kNN imputation ile dolduruluyor (5000 prob)...\n")
if (any(is.na(meth_filt4))) {
  na_count      <- sum(is.na(meth_filt4))
  na_probe_count <- sum(rowSums(is.na(meth_filt4)) > 0)
  cat(sprintf("    NA içeren hücre sayısı: %d (%d probda)\n", na_count, na_probe_count))
  imp_result <- impute::impute.knn(as.matrix(meth_filt4), k = 10)
  meth_step5 <- imp_result$data
  cat("    ✓ kNN imputation tamamlandı.\n")
  rm(imp_result); gc()
} else {
  cat("    NA bulunamadı, imputation atlandı.\n")
  meth_step5 <- as.matrix(meth_filt4)
}
rm(meth_filt4); gc()

# ---- Adım 7: Beta → M-value Dönüşümü ----
# Du et al. (2010): M-value = log2(beta / (1-beta))
# Beta sınırlama: 0.001 ve 0.999 (standart offset, Inf/NaN önler)
cat("  Adım 7: Beta → M-value dönüşümü [log2(beta/(1-beta))]...\n")
meth_step5[meth_step5 <= 0.001] <- 0.001
meth_step5[meth_step5 >= 0.999] <- 0.999
meth_mval <- log2(meth_step5 / (1 - meth_step5))
cat(sprintf("    M-value matrisi boyutu: %d prob x %d örnek\n",
            nrow(meth_mval), ncol(meth_mval)))
rm(meth_step5); gc()

# ---- Adım 8: Batch Effect Düzeltmesi (ComBat — TAM MATRİS) ----
# Batch değişkeni: TCGA barkodundan TSS (Tissue Source Site) kodu (6-7. karakter)
# Not: Xena verisinde plate ID (22-25. karakter) mevcut değildir.
# Uyarı: TSS, IDH mutasyonu ile kısmen confound olabilir.
# Çözüm: mod = model.matrix(~ Y_all) ile biyolojik sinyal korunur.
# Önemli: ComBat her zaman tam matris üzerinde tek seferde çalıştırılmalıdır;
# chunk'lama her parçanın kendi prior'ını tahmin etmesine yol açarak
# batch düzeltmesini istatistiksel olarak geçersiz kılar.
cat("  Adım 8: ComBat batch düzeltmesi uygulanıyor (tam matris, 5000 prob)...\n")

sample_names <- colnames(meth_mval)
batch_vec <- substr(sample_names, 6, 7)  # TSS kodu

# Singleton batch'leri birleştir (her batch'te en az 2 örnek olmalı)
batch_counts <- table(batch_vec)
small_batches <- names(batch_counts[batch_counts < 2])
if (length(small_batches) > 0) {
  cat(sprintf("    %d batch'te sadece 1 örnek bulundu. Bunlar 'Other' altında birleştiriliyor.\n", length(small_batches)))
  batch_vec[batch_vec %in% small_batches] <- "Other"
}

# Yeniden kontrol et, Other da tek kaldıysa en büyük batch ile birleştir
batch_counts <- table(batch_vec)
small_batches <- names(batch_counts[batch_counts < 2])
if (length(small_batches) > 0) {
  largest_batch <- names(which.max(batch_counts))
  batch_vec[batch_vec %in% small_batches] <- largest_batch
}

batch_fac <- as.factor(batch_vec)
cat(sprintf("    Tespit edilen batch sayısı: %d\n", nlevels(batch_fac)))

if (nlevels(batch_fac) > 1) {
  # mod matrisi: IDH durumunu koruyarak batch düzeltmesi yap
  mod_matrix <- model.matrix(~ Y_all)
  meth_combat <- tryCatch({
    sva::ComBat(dat = meth_mval, batch = batch_fac, mod = mod_matrix,
                par.prior = TRUE, prior.plots = FALSE)
  }, error = function(e) {
    cat("    Uyarı: Model matrisi ile ComBat hata verdi (olası confounding).\n")
    cat("    ComBat kovaryatsız çalıştırılıyor...\n")
    sva::ComBat(dat = meth_mval, batch = batch_fac, mod = NULL,
                par.prior = TRUE, prior.plots = FALSE)
  })
} else {
  cat("    Tek batch bulundu, ComBat atlandı.\n")
  meth_combat <- meth_mval
}
rm(meth_mval, batch_vec, batch_fac, sample_names); gc()

# ---- Adım 9: Transpose — mixOmics formatı: örnekler × problar ----

meth_final <- t(meth_combat)

X_mrna_test  <- mrna_final[test_idx, ]

# ---- Z-score: Train setinden fit, her iki sete uygula (data leakage önlemi) ----
# Bölüm 6 ve 7'de z-score kasıtlı bırakıldı; burada train istatistikleri
# (mean, sd) yalnızca train üzerinden hesaplanıp test setine aktarılıyor.
# scale() sütun bazlı çalışır → örnekler × özellikler formatında özellik bazında normalize eder.
cat("  Z-score normalizasyonu uygulanıyor (train-fit → test-transform)...\n")

mrna_means <- colMeans(X_mrna_train)
mrna_sds   <- apply(X_mrna_train, 2, sd)
mrna_sds[mrna_sds == 0] <- 1  # sıfır varyanslı gen kalmışsa sıfıra bölmeyi önle
X_mrna_train <- scale(X_mrna_train, center = mrna_means, scale = mrna_sds)
X_mrna_test  <- scale(X_mrna_test,  center = mrna_means, scale = mrna_sds)
rm(mrna_means, mrna_sds); gc()

meth_means <- colMeans(X_meth_train)
meth_sds   <- apply(X_meth_train, 2, sd)
meth_sds[meth_sds == 0] <- 1
X_meth_train <- scale(X_meth_train, center = meth_means, scale = meth_sds)
X_meth_test  <- scale(X_meth_test,  center = meth_means, scale = meth_sds)
rm(meth_means, meth_sds)
cat("  ✓ Z-score (train-fit) uygulandı: mRNA ve Metilasyon.\n\n")

# ---- 3. Confusion Matrix Grafiklerinin ggplot2 ile Çizimi ----
plot_confusion_matrix <- function(cm, title, filepath, accent_color) {
  cm_tbl <- as.data.frame(cm$table)
  cm_tbl$Prediction <- factor(cm_tbl$Prediction, levels = c("WT", "Mutant"))
  cm_tbl$Reference  <- factor(cm_tbl$Reference,  levels = c("Mutant", "WT"))

  midpoint <- mean(range(cm_tbl$Freq))

  p <- ggplot(cm_tbl, aes(x = Reference, y = Prediction, fill = Freq)) +
    geom_tile(color = "white", linewidth = 1.5) +
    geom_text(aes(label = Freq, color = ifelse(Freq > midpoint, "white", "black")),
              size = 8, fontface = "bold") +
    scale_color_identity() +
    scale_fill_gradient(low = "#EFF3FF", high = accent_color) +
    labs(title  = title,
         subtitle = "Test Cohort",
         x = "Gerçek IDH Durumu (Reference)",
         y = "Tahmin Edilen IDH Durumu (Prediction)") +
    theme_minimal(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", size = 13, hjust = 0.5),
      plot.subtitle = element_text(hjust = 0.5, size = 10, color = "gray40"),
      axis.title = element_text(face = "bold", size = 10),
      axis.text = element_text(face = "bold", size = 10),
      panel.grid = element_blank(),
      legend.position = "none"
    )

  ggsave(filepath, plot = p, width = 6, height = 5.5, dpi = 300)
}

  plot_confusion_matrix(cm_meth, "Confusion Matrix — sPLS-DA Methylation", file.path(output_dir, "CM_sPLSDA_Meth.png"), model_colors[2])
  plot_confusion_matrix(cm_diablo, "Confusion Matrix — DIABLO Multi-omics", file.path(output_dir, "CM_DIABLO.png"), model_colors[3])

# ---- 1. Bootstrap %95 Güven Aralıkları (ROC-AUC) ----
# Literatür: Carpenter & Bithell (2000) — 2000 stratified bootstrap önerilir
cat("  Bootstrap %95 CI hesaplanıyor (2000 resample)...\n")
set.seed(SEED_VAL)

ci_mrna   <- tryCatch(ci.auc(roc_mrna,   method = "bootstrap", boot.n = 2000, boot.stratified = TRUE), error = function(e) c(NA, NA, NA))
ci_meth   <- tryCatch(ci.auc(roc_meth,   method = "bootstrap", boot.n = 2000, boot.stratified = TRUE), error = function(e) c(NA, NA, NA))
ci_diablo <- tryCatch(ci.auc(roc_diablo, method = "bootstrap", boot.n = 2000, boot.stratified = TRUE), error = function(e) c(NA, NA, NA))
ci_rf     <- tryCatch(ci.auc(roc_rf,     method = "bootstrap", boot.n = 2000, boot.stratified = TRUE), error = function(e) c(NA, NA, NA))

ci_df <- data.frame(
  Model      = model_names,
  AUC        = sapply(roc_objects, function(r) round(as.numeric(auc(r)), 4)),
  CI_Lower   = round(c(ci_mrna[1], ci_meth[1], ci_diablo[1], ci_rf[1]), 4),
  CI_Upper   = round(c(ci_mrna[3], ci_meth[3], ci_diablo[3], ci_rf[3]), 4)
)
ci_df$CI_95 <- sprintf("%.4f (%.4f – %.4f)", ci_df$AUC, ci_df$CI_Lower, ci_df$CI_Upper)

cat("\n--- ROC-AUC Bootstrap %95 Güven Aralıkları ---\n")
print(ci_df[, c("Model", "CI_95")])

# Kaydet
write.csv(ci_df, file.path(output_dir, "Bootstrap_95CI_ROC_AUC.csv"), row.names = FALSE)
cat(sprintf("  ✓ Bootstrap CI tablosu kaydedildi: %s\n", file.path(output_dir, "Bootstrap_95CI_ROC_AUC.csv")))

# ---- 2. DeLong Testi — Model Çiftleri Arası AUC Farkı ----
# Literatür: DeLong et al. (1988) — iki ROC eğrisi arasındaki AUC farkının anlamlılığı
cat("\n  DeLong pairwise model karşılaştırma testleri...\n")

delong_pairs <- list(
  c("sPLS-DA mRNA", "sPLS-DA Meth"),
  c("sPLS-DA mRNA", "DIABLO"),
  c("sPLS-DA mRNA", "Concat RF"),
  c("sPLS-DA Meth", "DIABLO"),
  c("sPLS-DA Meth", "Concat RF"),
  c("DIABLO",       "Concat RF")
)

roc_named <- list(
  "sPLS-DA mRNA" = roc_mrna,
  "sPLS-DA Meth" = roc_meth,
  "DIABLO"       = roc_diablo,
  "Concat RF"    = roc_rf
)

delong_results <- data.frame(
  Model_1 = character(), Model_2 = character(),
  AUC_1 = numeric(), AUC_2 = numeric(),
  p_value = numeric(), Significant = character(),
  stringsAsFactors = FALSE
)

for (pair in delong_pairs) {
  test_result <- tryCatch({
    roc.test(roc_named[[pair[1]]], roc_named[[pair[2]]], method = "delong")
  }, error = function(e) NULL)

  if (!is.null(test_result)) {
    delong_results <- rbind(delong_results, data.frame(
      Model_1     = pair[1],
      Model_2     = pair[2],
      AUC_1       = round(as.numeric(auc(roc_named[[pair[1]]])), 4),
      AUC_2       = round(as.numeric(auc(roc_named[[pair[2]]])), 4),
      p_value     = signif(test_result$p.value, 4),
      Significant = ifelse(test_result$p.value < 0.05, "Evet *", "Hayır"),
      stringsAsFactors = FALSE
    ))
  }
}

cat("\n--- DeLong Pairwise AUC Karşılaştırma (p < 0.05 = Anlamlı Fark) ---\n")
print(delong_results)

# Kaydet
write.csv(delong_results, file.path(output_dir, "DeLong_Pairwise_AUC_Tests.csv"), row.names = FALSE)
cat(sprintf("  ✓ DeLong test tablosu kaydedildi: %s\n", file.path(output_dir, "DeLong_Pairwise_AUC_Tests.csv")))
