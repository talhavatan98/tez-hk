# HÜCRE 1 - Paket Kurulum
install.packages(c("UCSCXenaTools", "R.utils", "data.table",
                   "googledrive", "e1071", "pROC", "PRROC",
                   "glmnet", "BiocManager"))
BiocManager::install("mixOmics")

library(UCSCXenaTools)
library(R.utils)
library(data.table)
library(googledrive)

print("Tüm paketler hazır!")

# 1. İndir
xe <- XenaGenerate(subset = XenaCohorts == "TCGA lower grade glioma and glioblastoma (GBMLGG)")
xe_filter <- XenaFilter(xe, filterDatasets = "HiSeqV2$|HumanMethylation450$|GBMLGG_clinicalMatrix$")
xe_query <- XenaQuery(xe_filter)
XenaDownload(xe_query, destdir = "/content")

# 2. Oku
klinik <- fread("/content/TCGA.GBMLGG.sampleMap/GBMLGG_clinicalMatrix", data.table = FALSE)
rna    <- fread("/content/TCGA.GBMLGG.sampleMap/HiSeqV2.gz", data.table = FALSE)
met    <- fread("/content/TCGA.GBMLGG.sampleMap/HumanMethylation450.gz", data.table = FALSE)

# 3. Ortak hastalar
ortak_hastalar <- sort(intersect(intersect(klinik$sampleID, colnames(rna)[-1]), colnames(met)[-1]))
print(paste("Ortak hasta:", length(ortak_hastalar)))

# 4. Hizala ve transpoze et
klinik_temiz <- klinik[match(ortak_hastalar, klinik$sampleID), ]
rownames(rna) <- rna[, 1]
rna_t <- as.data.frame(t(rna[, ortak_hastalar]))
rownames(met) <- met[, 1]
met_t <- as.data.frame(t(met[, ortak_hastalar]))

# 5. RAM temizle
rm(klinik, rna, met)
gc()

# 6. Y_IDH — MC3'ten oluştur
xe_gbmlgg <- XenaGenerate(subset = XenaCohorts == "TCGA lower grade glioma and glioblastoma (GBMLGG)")
xe_mut_filter <- XenaFilter(xe_gbmlgg, filterDatasets = "mc3_gene_level/GBMLGG_mc3_gene_level.txt")
XenaDownload(XenaQuery(xe_mut_filter), destdir = "/content")

mut_dosya <- list.files("/content/mc3_gene_level", full.names = TRUE)[1]
mutasyon_verisi <- fread(mut_dosya, data.table = FALSE)
mutasyon_verisi <- mutasyon_verisi[!is.na(mutasyon_verisi[, 1]), ]
rownames(mutasyon_verisi) <- mutasyon_verisi[, 1]
mutasyon_numeric <- mutasyon_verisi[, -1]

eslesenler <- intersect(ortak_hastalar, colnames(mutasyon_numeric))
idh_matris <- mutasyon_numeric[c("IDH1", "IDH2"), eslesenler]
idh_matris[is.na(idh_matris)] <- 0

Y_vektor <- setNames(rep("NO", length(ortak_hastalar)), ortak_hastalar)
idh_skoru <- apply(idh_matris, 2, function(x) if(any(x == 1)) "YES" else "NO")
Y_vektor[names(idh_skoru)] <- idh_skoru
Y_IDH <- as.factor(Y_vektor)

print("Y_IDH dağılımı:")
print(table(Y_IDH))

rm(mutasyon_verisi, mutasyon_numeric, idh_matris)
gc()

print("Hücre 2 tamamlandı!")

# 3. hücre, ÖNCE TRAIN/TEST SPLIT (Hiçbir işlem yapmadan bölüyoruz)
if(!require("caret")) install.packages("caret")
library(caret)

set.seed(123)
train_idx <- createDataPartition(Y_IDH, p = 0.70, list = FALSE)

Y_train <- Y_IDH[train_idx]
Y_test  <- Y_IDH[-train_idx]
rna_train <- rna_t[train_idx, ]
rna_test  <- rna_t[-train_idx, ]
met_train <- met_t[train_idx, ]
met_test  <- met_t[-train_idx, ]

# Hafızayı rahatlatalım
rm(rna_t, met_t)
gc()

print(paste("Split Tamam! Train:", nrow(rna_train), "| Test:", nrow(rna_test)))

# 2. METİLASYON NA FİLTRESİ (SADECE Train'e göre hesaplanır)
prob_na_oran_train <- colMeans(is.na(met_train))
# Train'de %20'den fazla NA olan probları atıyoruz
gecerli_problar <- names(prob_na_oran_train[prob_na_oran_train < 0.20])

met_train <- met_train[, gecerli_problar]
met_test  <- met_test[, gecerli_problar]
print(paste("NA filtresi sonrası kalan prob:", ncol(met_train)))

# 3. NA DOLDURMA (SADECE Train ortalamaları ile)
# Sızıntıyı önlemek için ortalamalar sadece Train setinden alınır!
train_means <- colMeans(met_train, na.rm = TRUE)

# Train içindeki NA'leri doldur
for(i in which(colSums(is.na(met_train)) > 0)) {
  met_train[is.na(met_train[, i]), i] <- train_means[i]
}

# Test içindeki NA'leri AYNI Train ortalamaları ile doldur (Sızıntı bitti!)
for(i in which(colSums(is.na(met_test)) > 0)) {
  met_test[is.na(met_test[, i]), i] <- train_means[i]
}

print(paste("Doldurma sonrası Train NA:", sum(is.na(met_train))))
print(paste("Doldurma sonrası Test NA:", sum(is.na(met_test))))

print("Hücre 3 (Sıfır Veri Sızıntısı ile) tamamlandı!")

# 1. Varyans filtresi (sadece train)
rna_varyans <- apply(rna_train, 2, var)
top_rna_genler <- names(sort(rna_varyans, decreasing = TRUE)[1:1000])
rna_train_filtre <- rna_train[, top_rna_genler]
rna_test_filtre  <- rna_test[, top_rna_genler]

met_varyans <- apply(met_train, 2, var)
top_met_problar <- names(sort(met_varyans, decreasing = TRUE)[1:5000])
met_train_filtre <- met_train[, top_met_problar]
met_test_filtre  <- met_test[, top_met_problar]

rm(rna_train, rna_test, met_train, met_test)
gc()

# 2. Z-score scaling (train'e fit, test'e uygula)
rna_means <- colMeans(rna_train_filtre)
rna_sds   <- apply(rna_train_filtre, 2, sd)
rna_train_scaled <- as.matrix(scale(rna_train_filtre, center = rna_means, scale = rna_sds))
rna_test_scaled  <- as.matrix(scale(rna_test_filtre,  center = rna_means, scale = rna_sds))

met_means <- colMeans(met_train_filtre)
met_sds   <- apply(met_train_filtre, 2, sd)
met_train_scaled <- as.matrix(scale(met_train_filtre, center = met_means, scale = met_sds))
met_test_scaled  <- as.matrix(scale(met_test_filtre,  center = met_means, scale = met_sds))

# DİKKAT: _filtre uzantılı ham dosyaları SİLMİYORUZ, listeyi ona göre güncelledik.
rm(rna_means, rna_sds, met_means, met_sds, rna_varyans, met_varyans)
gc()

print(paste("RNA train scaled:", nrow(rna_train_scaled), "x", ncol(rna_train_scaled)))
print(paste("Met train scaled:", nrow(met_train_scaled), "x", ncol(met_train_scaled)))
print(paste("RNA NA:", sum(is.na(rna_train_scaled))))
print(paste("Met NA:", sum(is.na(met_train_scaled))))

print("Hücre 4 tamamlandı!")

library(googledrive)
drive_auth()

save(rna_train_scaled, rna_test_scaled,
     met_train_scaled, met_test_scaled,
     # Scale edilmemiş ama filtrelenmiş yedekler (mixOmics vb. için)
     rna_train_filtre, rna_test_filtre,
     met_train_filtre, met_test_filtre,
     Y_train, Y_test, Y_IDH,
     top_rna_genler, top_met_problar,
     klinik_temiz, ortak_hastalar, train_idx,
     file = "/content/tez_veri_final.RData")

drive_upload("/content/tez_veri_final.RData",
             name = "tez_veri_final.RData",
             overwrite = TRUE)

print("Yedekleme tam liste olarak başarıyla tamamlandı!")
print(ls())

silinecekler <- c("eslesenler", "gecerli_problar", "i", "idh_skoru", "mut_dosya",
                  "prob_na_oran_train", "train_means", "xe", "xe_filter",
                  "xe_gbmlgg", "xe_mut_filter", "xe_query", "Y_vektor")
rm(list = intersect(ls(), silinecekler))
gc()
print("Geçici ön işleme değişkenleri başarıyla temizlendi, ana veriler GÜVENDE!")
