# Matthews Correlation Coefficient (MCC) — caret doğrudan vermez
calc_mcc <- function(cm) {
  tp <- cm$table[1, 1]  # Predicted Mutant, Actual Mutant
  tn <- cm$table[2, 2]  # Predicted WT, Actual WT
  fp <- cm$table[1, 2]  # Predicted Mutant, Actual WT
  fn <- cm$table[2, 1]  # Predicted WT, Actual Mutant
  denom <- sqrt(as.numeric(tp + fp) * as.numeric(tp + fn) *
                as.numeric(tn + fp) * as.numeric(tn + fn))
  if (denom == 0) return(0)
  return((tp * tn - fp * fn) / denom)
}

# Permütasyon testi — Y_test etiketleri N_PERM kez karıştırılır, model sabit tutulur.
# p-değeri: null_AUC >= gözlenen_AUC olan permütasyon oranı.
run_permutation_test <- function(obs_auc, prob_scores, Y_labels, n_perm, seed = SEED_VAL) {
  set.seed(seed)
  null_aucs <- vapply(seq_len(n_perm), function(i) {
    Y_perm <- sample(Y_labels)
    tryCatch(
      as.numeric(auc(roc(Y_perm, prob_scores,
                         levels = c("WT", "Mutant"), direction = "<", quiet = TRUE))),
      error = function(e) 0.5
    )
  }, numeric(1))
  list(p_value = mean(null_aucs >= obs_auc), null_mean = mean(null_aucs))
}

# Not: Xena verisinde plate ID (22-25. karakter) mevcut değildir.

# Çözüm: mod = model.matrix(~ Y_all) ile biyolojik sinyal korunur.
# Önemli: ComBat her zaman tam matris üzerinde tek seferde çalıştırılmalıdır;
# chunk'lama her parçanın kendi prior'ını tahmin etmesine yol açarak
# batch düzeltmesini istatistiksel olarak geçersiz kılar.

# Singleton batch'leri birleştir (her batch'te en az 2 örnek olmalı)

small_batches <- names(batch_counts[batch_counts < 2])

  cat(sprintf("    %d batch'te sadece 1 örnek bulundu. Bunlar 'Other' altında birleştiriliyor.\n", length(small_batches)))
  batch_vec[batch_vec %in% small_batches] <- "Other"
}

# Yeniden kontrol et, Other da tek kaldıysa en büyük batch ile birleştir
batch_counts <- table(batch_vec)
small_batches <- names(batch_counts[batch_counts < 2])
if (length(small_batches) > 0) {
  largest_batch <- names(which.max(batch_counts))
  batch_vec[batch_vec %in% small_batches] <- largest_batch
}

batch_fac <- as.factor(batch_vec)
cat(sprintf("    Tespit edilen batch sayısı: %d\n", nlevels(batch_fac)))

if (nlevels(batch_fac) > 1) {
  # mod matrisi: IDH durumunu koruyarak batch düzeltmesi yap
  mod_matrix <- model.matrix(~ Y_all)
  meth_combat <- tryCatch({
    sva::ComBat(dat = meth_mval, batch = batch_fac, mod = mod_matrix,
                par.prior = TRUE, prior.plots = FALSE)
  }, error = function(e) {
    cat("    Uyarı: Model matrisi ile ComBat hata verdi (olası confounding).\n")
    cat("    ComBat kovaryatsız çalıştırılıyor...\n")
    sva::ComBat(dat = meth_mval, batch = batch_fac, mod = NULL,
                par.prior = TRUE, prior.plots = FALSE)
  })
} else {

# ---- Z-score: Train setinden fit, her iki sete uygula (data leakage önlemi) ----

plot_confusion_matrix <- function(cm, title, filepath, accent_color) {
  cm_tbl <- as.data.frame(cm$table)
  cm_tbl$Prediction <- factor(cm_tbl$Prediction, levels = c("WT", "Mutant"))
  cm_tbl$Reference  <- factor(cm_tbl$Reference,  levels = c("Mutant", "WT"))

  midpoint <- mean(range(cm_tbl$Freq))

  p <- ggplot(cm_tbl, aes(x = Reference, y = Prediction, fill = Freq)) +
    geom_tile(color = "white", linewidth = 1.5) +
    geom_text(aes(label = Freq, color = ifelse(Freq > midpoint, "white", "black")),
              size = 8, fontface = "bold") +
    scale_color_identity() +
    scale_fill_gradient(low = "#EFF3FF", high = accent_color) +
    labs(title  = title,
         subtitle = "Test Cohort",
         x = "Gerçek IDH Durumu (Reference)",
         y = "Tahmin Edilen IDH Durumu (Prediction)") +
    theme_minimal(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", size = 13, hjust = 0.5),
      plot.subtitle = element_text(hjust = 0.5, size = 10, color = "gray40"),
      axis.title = element_text(face = "bold", size = 10),
      axis.text = element_text(face = "bold", size = 10),
      panel.grid = element_blank(),
      legend.position = "none"
    )

  ggsave(filepath, plot = p, width = 6, height = 5.5, dpi = 300)
  print(p)  # Colab inline görüntü
}

# ---- 2. DeLong Testi — Model Çiftleri Arası AUC Farkı ----
# Literatür: DeLong et al. (1988) — iki ROC eğrisi arasındaki AUC farkının anlamlılığı
cat("\n  DeLong pairwise model karşılaştırma testleri...\n")

delong_pairs <- list(
  c("sPLS-DA mRNA", "sPLS-DA Meth"),
  c("sPLS-DA mRNA", "DIABLO"),
  c("sPLS-DA mRNA", "Concat RF"),
  c("sPLS-DA Meth", "DIABLO"),
  c("sPLS-DA Meth", "Concat RF"),
  c("DIABLO",       "Concat RF")
)

roc_named <- list(
  "sPLS-DA mRNA" = roc_mrna,
  "sPLS-DA Meth" = roc_meth,
  "DIABLO"       = roc_diablo,
  "Concat RF"    = roc_rf
)

delong_results <- data.frame(
  Model_1 = character(), Model_2 = character(),
  Delta_AUC = numeric(),
  CI_Lower = numeric(), CI_Upper = numeric(),
  Z_stat = numeric(),
  DeLong_p = numeric(),
  Bootstrap_p = numeric(),
  Significant = character(),
  stringsAsFactors = FALSE
)

for (pair in delong_pairs) {
  dl <- tryCatch(
    roc.test(roc_named[[pair[1]]], roc_named[[pair[2]]], method = "delong",     conf.level = 0.95),
    error = function(e) NULL
  )
  bs <- tryCatch(
    roc.test(roc_named[[pair[1]]], roc_named[[pair[2]]], method = "bootstrap",  boot.n = 2000, boot.stratified = TRUE),
    error = function(e) NULL
  )

  if (!is.null(dl)) {
    delta   <- round(as.numeric(auc(roc_named[[pair[1]]])) - as.numeric(auc(roc_named[[pair[2]]])), 4)
    ci_lo   <- if (!is.null(dl$conf.int)) round(dl$conf.int[1], 4) else NA
    ci_hi   <- if (!is.null(dl$conf.int)) round(dl$conf.int[2], 4) else NA
    z_val   <- round(as.numeric(dl$statistic), 4)
    dl_p    <- signif(dl$p.value, 4)
    bs_p    <- if (!is.null(bs)) signif(bs$p.value, 4) else NA
    sig     <- ifelse(!is.na(dl_p) & dl_p < 0.05, "Evet *", "Hayır")

    delong_results <- rbind(delong_results, data.frame(
      Model_1     = pair[1],
      Model_2     = pair[2],
      Delta_AUC   = delta,
      CI_Lower    = ci_lo,
      CI_Upper    = ci_hi,
      Z_stat      = z_val,
      DeLong_p    = dl_p,
      Bootstrap_p = bs_p,
      Significant = sig,
      stringsAsFactors = FALSE
    ))
  }
}

map_probes_to_genes <- function(probe_ids, annotation) {
  # UCSC_RefGene_Name sütununda birden fazla gen ";" ile ayrılmış olabilir
  valid_probes <- probe_ids[probe_ids %in% rownames(annotation)]
  if (length(valid_probes) == 0) return(character(0))
  raw   <- annotation[valid_probes, "UCSC_RefGene_Name"]
  genes <- unique(unlist(strsplit(raw[!is.na(raw) & raw != ""], ";")))
  sort(genes)
}

            age_col, sex_col, grade_col, os_col, event_col))

# ---- Yardımcı: grup bazlı özet ----
grp <- clinical_df$IDH_binary  # "Mutant" / "WT"
mutant_idx <- grp == "Mutant"
wt_idx     <- grp == "WT"
n_total  <- nrow(clinical_df)
n_mutant <- sum(mutant_idx)
n_wt     <- sum(wt_idx)

demo_rows <- list()

# ---- Yaş ----
if (!is.na(age_col)) {
  age_vals <- suppressWarnings(as.numeric(clinical_df[[age_col]]))
  age_test <- tryCatch(t.test(age_vals[mutant_idx], age_vals[wt_idx]), error = function(e) NULL)
  age_p    <- if (!is.null(age_test)) signif(age_test$p.value, 4) else NA
  demo_rows[["Yaş (Yıl), Ort. ± SS"]] <- data.frame(
    Degisken   = "Yaş (Yıl), Ort. ± SS",
    Toplam     = sprintf("%.1f ± %.1f", mean(age_vals, na.rm=TRUE), sd(age_vals, na.rm=TRUE)),
    IDH_Mutant = sprintf("%.1f ± %.1f", mean(age_vals[mutant_idx], na.rm=TRUE), sd(age_vals[mutant_idx], na.rm=TRUE)),
    IDH_WT     = sprintf("%.1f ± %.1f", mean(age_vals[wt_idx], na.rm=TRUE), sd(age_vals[wt_idx], na.rm=TRUE)),
    P_Degeri   = as.character(age_p),
    stringsAsFactors = FALSE
  )

  # ---- Yaş Grupları (<40 / 40-60 / >60) ----
  age_grp <- cut(age_vals, breaks = c(-Inf, 39.99, 60, Inf),
                 labels = c("< 40", "40 - 60", "> 60"), right = TRUE)
  age_grp_tbl  <- table(age_grp, grp)
  age_grp_test <- tryCatch(
    chisq.test(age_grp_tbl),
    error = function(e) tryCatch(
      fisher.test(age_grp_tbl, simulate.p.value = TRUE, B = 10000),
      error = function(e2) NULL
    )
  )
  age_grp_p <- if (!is.null(age_grp_test)) signif(age_grp_test$p.value, 4) else NA

  demo_rows[["Yaş Grupları (başlık)"]] <- data.frame(
    Degisken = "Yaş Grupları, n (%)", Toplam = "", IDH_Mutant = "", IDH_WT = "",
    P_Degeri = as.character(age_grp_p), stringsAsFactors = FALSE
  )
  for (ag in c("< 40", "40 - 60", "> 60")) {
    ng_total  <- sum(age_grp == ag, na.rm = TRUE)
    ng_mutant <- sum(age_grp[mutant_idx] == ag, na.rm = TRUE)
    ng_wt     <- sum(age_grp[wt_idx]     == ag, na.rm = TRUE)
    demo_rows[[paste0("YasGrp_", ag)]] <- data.frame(
      Degisken   = sprintf("  %s", ag),
      Toplam     = sprintf("%d (%%%0.1f)", ng_total,  100 * ng_total  / n_total),
      IDH_Mutant = sprintf("%d (%%%0.1f)", ng_mutant, 100 * ng_mutant / n_mutant),
      IDH_WT     = sprintf("%d (%%%0.1f)", ng_wt,     100 * ng_wt     / n_wt),
      P_Degeri   = "",
      stringsAsFactors = FALSE
    )
  }
}

# ---- Cinsiyet ----
if (!is.na(sex_col)) {
  sex_vals <- tolower(as.character(clinical_df[[sex_col]]))
  male_total   <- sum(sex_vals %in% c("male", "erkek"), na.rm=TRUE)
  male_mutant  <- sum(sex_vals[mutant_idx] %in% c("male", "erkek"), na.rm=TRUE)
  male_wt      <- sum(sex_vals[wt_idx]     %in% c("male", "erkek"), na.rm=TRUE)
  sex_tbl      <- table(sex_vals, grp)
  sex_test     <- tryCatch(fisher.test(sex_tbl, simulate.p.value = TRUE, B = 10000), error = function(e) NULL)
  sex_p        <- if (!is.null(sex_test)) signif(sex_test$p.value, 4) else NA
  demo_rows[["Cinsiyet (Erkek)"]] <- data.frame(
    Degisken   = "Cinsiyet (Erkek), n (%)",
    Toplam     = sprintf("%d (%%%0.1f)", male_total,  100*male_total/n_total),
    IDH_Mutant = sprintf("%d (%%%0.1f)", male_mutant, 100*male_mutant/n_mutant),
    IDH_WT     = sprintf("%d (%%%0.1f)", male_wt,     100*male_wt/n_wt),
    P_Degeri   = as.character(sex_p),
    stringsAsFactors = FALSE
  )
}

# ---- Grade ----
if (!is.na(grade_col)) {
  grade_vals <- toupper(as.character(clinical_df[[grade_col]]))
  # TCGA GBMLGG'de GBM örnekleri neoplasm_histologic_grade sütununda boş bırakılır.
  # _primary_disease sütunundaki "gbm" etiketiyle bu örnekleri G4 olarak doldur.
  if ("_primary_disease" %in% colnames(clinical_df)) {
    disease_vals <- tolower(as.character(clinical_df[["_primary_disease"]]))
    is_gbm <- grepl("glioblastoma|gbm", disease_vals)
    grade_vals[is_gbm & (is.na(grade_vals) | grade_vals == "")] <- "G4"
  }
  # Sadece G2/G3/G4 tut; [Discrepancy], GX gibi anlamlı olmayan kategorileri çıkar
  valid_grades <- c("G2", "G3", "G4")
  grade_vals[!(grade_vals %in% valid_grades)] <- NA
  grade_levels <- c("G2", "G3", "G4")
  grade_tbl  <- table(grade_vals, grp)
  grade_test <- tryCatch(fisher.test(grade_tbl, simulate.p.value = TRUE, B = 10000), error = function(e) NULL)
  grade_p    <- if (!is.null(grade_test)) signif(grade_test$p.value, 4) else NA
  # Başlık satırı
  demo_rows[["Grade (başlık)"]] <- data.frame(
    Degisken = "Derece (Grade), n (%)", Toplam = "", IDH_Mutant = "", IDH_WT = "",
    P_Degeri = as.character(grade_p), stringsAsFactors = FALSE
  )
  for (g in grade_levels) {
    ng_total  <- sum(grade_vals == g, na.rm=TRUE)
    ng_mutant <- sum(grade_vals[mutant_idx] == g, na.rm=TRUE)
    ng_wt     <- sum(grade_vals[wt_idx]     == g, na.rm=TRUE)
    demo_rows[[paste0("Grade_", g)]] <- data.frame(
      Degisken   = sprintf("  %s", g),
      Toplam     = sprintf("%d (%%%0.1f)", ng_total,  100*ng_total/n_total),
      IDH_Mutant = sprintf("%d (%%%0.1f)", ng_mutant, 100*ng_mutant/n_mutant),
      IDH_WT     = sprintf("%d (%%%0.1f)", ng_wt,     100*ng_wt/n_wt),
      P_Degeri   = "",
      stringsAsFactors = FALSE
    )
  }
}

# ---- Vefat / Vital Status ----
if (!is.na(event_col)) {
  ev_raw  <- clinical_df[[event_col]]
  # 1/0 veya "Dead"/"Alive" formatını normalize et
  if (is.numeric(ev_raw)) {
    ev_vals <- as.integer(ev_raw)
  } else {
    ev_vals <- ifelse(tolower(as.character(ev_raw)) %in% c("dead","deceased","1"), 1L, 0L)
  }
  died_total  <- sum(ev_vals == 1, na.rm=TRUE)
  died_mutant <- sum(ev_vals[mutant_idx] == 1, na.rm=TRUE)
  died_wt     <- sum(ev_vals[wt_idx]     == 1, na.rm=TRUE)
  ev_test     <- tryCatch(chisq.test(table(ev_vals, grp)), error = function(e) NULL)
  ev_p        <- if (!is.null(ev_test)) signif(ev_test$p.value, 4) else NA
  demo_rows[["Vefat"]] <- data.frame(
    Degisken   = "Vefat Eden, n (%)",
    Toplam     = sprintf("%d (%%%0.1f)", died_total,  100*died_total/n_total),
    IDH_Mutant = sprintf("%d (%%%0.1f)", died_mutant, 100*died_mutant/n_mutant),
    IDH_WT     = sprintf("%d (%%%0.1f)", died_wt,     100*died_wt/n_wt),
    P_Degeri   = as.character(ev_p),
    stringsAsFactors = FALSE
  )
}

# ---- Medyan Sağkalım ----
if (!is.na(os_col) && !is.na(event_col)) {
  # OS süresi: ölen → days_to_death, yaşayan → days_to_last_followup
  followup_col_med <- intersect(c("days_to_last_followup", "days_to_last_known_alive"), colnames(clinical_df))[1]
  days_death_med   <- suppressWarnings(as.numeric(clinical_df[[os_col]]))
  days_fu_med      <- if (!is.na(followup_col_med)) suppressWarnings(as.numeric(clinical_df[[followup_col_med]])) else rep(NA_real_, nrow(clinical_df))
  os_time_med      <- ifelse(!is.na(days_death_med), days_death_med, days_fu_med)
  os_month <- os_time_med / 30.44
  ev_vals2 <- if (is.numeric(clinical_df[[event_col]])) {
    as.integer(clinical_df[[event_col]])
  } else {
    ifelse(tolower(as.character(clinical_df[[event_col]])) %in% c("dead","deceased","1"), 1L, 0L)
  }
  sf_total  <- tryCatch(survfit(Surv(os_month, ev_vals2) ~ 1),                    error=function(e) NULL)
  sf_mutant <- tryCatch(survfit(Surv(os_month[mutant_idx], ev_vals2[mutant_idx]) ~ 1), error=function(e) NULL)
  sf_wt     <- tryCatch(survfit(Surv(os_month[wt_idx],     ev_vals2[wt_idx])     ~ 1), error=function(e) NULL)
  get_median <- function(sf) if (!is.null(sf)) round(summary(sf)$table["median"], 1) else NA
  demo_rows[["Medyan Saglıkalım"]] <- data.frame(
    Degisken   = "Medyan Sağkalım (Ay)",
    Toplam     = as.character(get_median(sf_total)),
    IDH_Mutant = as.character(get_median(sf_mutant)),
    IDH_WT     = as.character(get_median(sf_wt)),
    P_Degeri   = "-",
    stringsAsFactors = FALSE
  )
}

# ---- Kaplan-Meier Eğrileri ----
if (!is.na(os_col) && !is.na(event_col)) {
  tryCatch({
    # OS süresi: ölen → days_to_death, yaşayan (sansür) → days_to_last_followup
    # days_to_death tek başına kullanılırsa yaşayan hastalar NA olur ve KM'den düşer
    followup_col <- intersect(c("days_to_last_followup", "days_to_last_known_alive"), colnames(clinical_df))[1]
    days_death   <- suppressWarnings(as.numeric(clinical_df[[os_col]]))
    days_followup <- if (!is.na(followup_col)) suppressWarnings(as.numeric(clinical_df[[followup_col]])) else rep(NA_real_, nrow(clinical_df))
    os_time_days <- ifelse(!is.na(days_death), days_death, days_followup)

    km_df <- data.frame(
      time   = os_time_days / 30.44,
      event  = ev_vals2,
      group  = grp
    )
    km_df <- km_df[!is.na(km_df$time) & km_df$time > 0 & !is.na(km_df$event), ]

    km_fit    <- survfit(Surv(time, event) ~ group, data = km_df)
    km_logrank <- survdiff(Surv(time, event) ~ group, data = km_df)
    km_p      <- signif(1 - pchisq(km_logrank$chisq, df = 1), 4)

    km_plot <- ggsurvplot(
      km_fit,
      data          = km_df,
      palette       = c("#D7191C", "#2C7BB6"),
      conf.int      = TRUE,
      pval          = TRUE,
      pval.method   = TRUE,
      risk.table    = TRUE,
      risk.table.height = 0.25,
      legend.labs   = c("IDH Mutant", "IDH Yabanıl Tip"),
      legend.title  = "IDH Durumu",
      xlab          = "Süre (Ay)",
      ylab          = "Genel Sağkalım Olasılığı",
      title         = "Kaplan-Meier — Genel Sağkalım (IDH Mutant vs WT)",
      ggtheme       = theme_minimal(base_size = 12)
    )

    # Kaydet
    png(file.path(output_dir, "KM_Overall_Survival_IDH.png"),
        width = 2400, height = 2400, res = 300)
    print(km_plot)
    dev.off()

    # Colab inline
    print(km_plot)
    cat(sprintf("  ✓ Kaplan-Meier eğrisi kaydedildi: KM_Overall_Survival_IDH.png\n"))
    cat(sprintf("  Log-rank p-değeri: %s\n", km_p))
  }, error = function(e) cat("  KM eğrisi çizilemedi:", e$message, "\n"))
}
